# 在线人员统计使用示例（内存版 MVP）

本文档演示如何使用会议在线人员功能，包括 WebSocket 接入与 HTTP 查询接口。

## WebSocket 接入（Presence）

- 路径：`/api/meetings/ws/meetings/{meeting_id}/presence`
- 认证：在 `Authorization: Bearer <access_token>` 或 `?token=<access_token>` 传入 access token
- 心跳：客户端每 20–30s 发送一次心跳消息 `{ "type": "heartbeat" }`

示例（浏览器原生 WebSocket）：

```js
const meetingId = "<会议ID>";
const token = "<access_token>"; // 登录后获取
const ws = new WebSocket(`ws://localhost:8000/api/meetings/ws/meetings/${meetingId}/presence?token=${token}`);

ws.onopen = () => {
  console.log("Presence connected");
  // 定时心跳
  setInterval(() => {
    ws.send(JSON.stringify({ type: "heartbeat" }));
  }, 30000);
};

ws.onmessage = (event) => {
  const msg = JSON.parse(event.data);
  if (msg.type === "presence.join") {
    console.log("用户加入：", msg.user, "在线人数：", msg.online_count);
  } else if (msg.type === "presence.leave") {
    console.log("用户离开：", msg.user, "在线人数：", msg.online_count);
  } else if (msg.type === "presence.snapshot") {
    console.log("当前在线用户：", msg.users);
  } else if (msg.type === "presence.heartbeat_ack") {
    // 心跳回执，可按需忽略
  }
};

ws.onclose = () => {
  console.log("Presence disconnected");
};
```

发送快照请求（可选）：

```js
ws.send(JSON.stringify({ type: "snapshot.request" }));
```

服务端事件格式：

```json
// 用户加入
{
  "type": "presence.join",
  "meeting_id": "...",
  "user": { "id": "...", "name": "张三", "role": "user" },
  "online_count": 3
}

// 用户离开
{
  "type": "presence.leave",
  "meeting_id": "...",
  "user": { "id": "...", "name": "张三", "role": "user" },
  "online_count": 2
}

// 快照
{
  "type": "presence.snapshot",
  "meeting_id": "...",
  "online_count": 2,
  "users": [
    { "id": "...", "last_active": "2025-01-01T12:00:00Z", "metadata": { "name": "张三", "role": "user" } },
    { "id": "...", "last_active": "2025-01-01T12:00:10Z", "metadata": { "name": "李四", "role": "admin" } }
  ]
}
```

## HTTP 查询接口

均需在 `Authorization: Bearer <access_token>` 头部带上 access token。

- 查询在线人数：`GET /api/meetings/{meeting_id}/online-count`

```bash
curl -H "Authorization: Bearer <access_token>" \
  "http://localhost:8000/api/meetings/<会议ID>/online-count"
```

响应：

```json
{ "meeting_id": "...", "online_count": 3 }
```

- 查询在线用户列表：`GET /api/meetings/{meeting_id}/online-users`

```bash
curl -H "Authorization: Bearer <access_token>" \
  "http://localhost:8000/api/meetings/<会议ID>/online-users"
```

响应：

```json
{
  "meeting_id": "...",
  "online_users": [
    { "id": "...", "last_active": "2025-01-01T12:00:00Z", "metadata": { "name": "张三", "role": "user" } }
  ]
}
```

## 前端对接示例（Vue 3，核心代码）

```ts
// 构建 Presence WebSocket URL（前端开发环境可通过代理转发到后端）
function buildWsUrl(meetingId: string, token: string) {
  const scheme = location.protocol === 'https:' ? 'wss' : 'ws';
  const host = location.host; // 例：localhost:5173，开发阶段由代理转发至 8000
  return `${scheme}://${host}/api/meetings/ws/meetings/${encodeURIComponent(meetingId)}/presence?token=${encodeURIComponent(token)}`;
}

let ws: WebSocket | null = null;
let heartbeatTimer: number | undefined;
let wsState: 'idle' | 'connecting' | 'connected' = 'idle';
let onlineCount = 0;
let onlineUsers: Array<{ id: string; last_active?: string; metadata?: any }> = [];

function sendEvent(payload: any) {
  try { ws?.send(JSON.stringify(payload)); } catch {}
}

function joinPresence(meetingId: string, token: string) {
  if (wsState !== 'idle') return;
  wsState = 'connecting';
  ws = new WebSocket(buildWsUrl(meetingId, token));

  ws.onopen = () => {
    wsState = 'connected';
    // 请求快照
    sendEvent({ type: 'snapshot' });
    // 启动心跳（20s）
    heartbeatTimer = window.setInterval(() => sendEvent({ type: 'heartbeat' }), 20000);
  };

  ws.onmessage = (ev) => {
    const msg = JSON.parse(ev.data || '{}');
    const type = (msg.type || '').toLowerCase();
    if (type === 'presence.snapshot') {
      onlineCount = msg.online_count ?? 0;
      onlineUsers = Array.isArray(msg.users) ? msg.users : [];
    } else if (type === 'presence.join') {
      onlineCount = msg.online_count ?? onlineCount;
      const u = msg.user || {};
      const id = u.id || u.user_id;
      if (id) {
        const item = { id, last_active: new Date().toISOString(), metadata: { name: u.name, role: u.role } };
        const idx = onlineUsers.findIndex(x => String(x.id) === String(id));
        if (idx === -1) onlineUsers.push(item); else onlineUsers[idx] = { ...onlineUsers[idx], ...item };
      }
    } else if (type === 'presence.leave') {
      onlineCount = msg.online_count ?? onlineCount;
      const u = msg.user || {};
      const id = u.id || u.user_id;
      if (id) onlineUsers = onlineUsers.filter(x => String(x.id) !== String(id));
    }
  };

  ws.onclose = () => { cleanupWs(); startPolling(meetingId, token); };
  ws.onerror = () => { cleanupWs(); startPolling(meetingId, token); };
}

function leavePresence(meetingId: string, token: string) {
  cleanupWs();
  wsState = 'idle';
  // 轻微延迟，确保服务端完成离线清理
  setTimeout(() => refreshOnline(meetingId, token), 200);
  startPolling(meetingId, token);
}

function cleanupWs() {
  if (heartbeatTimer) { clearInterval(heartbeatTimer); heartbeatTimer = undefined; }
  try { ws?.close(); } catch {}
  ws = null;
  wsState = 'idle';
}

// 未连接时的定时轮询（HTTP）
const POLL_INTERVAL_MS = 5000;
let pollTimer: number | undefined;

async function refreshOnline(meetingId: string, token: string) {
  const headers = { Authorization: `Bearer ${token}` };
  const [countRes, usersRes] = await Promise.all([
    fetch(`/api/meetings/${encodeURIComponent(meetingId)}/online-count`, { headers }),
    fetch(`/api/meetings/${encodeURIComponent(meetingId)}/online-users`, { headers }),
  ]);
  const countData = await countRes.json();
  const usersData = await usersRes.json();
  onlineCount = countData?.online_count ?? 0;
  onlineUsers = Array.isArray(usersData?.online_users) ? usersData.online_users : [];
}

function startPolling(meetingId: string, token: string) {
  stopPolling();
  pollTimer = window.setInterval(() => {
    if (wsState !== 'connected') refreshOnline(meetingId, token);
  }, POLL_INTERVAL_MS);
}

function stopPolling() {
  if (pollTimer) { clearInterval(pollTimer); pollTimer = undefined; }
}
```

## 说明

- 在线人数以“用户维度”计数，用户多设备登录只计 1 人。
- 当前实现为内存版（单实例），不支持多进程/多实例合并统计；后续可扩展至 Redis 版本。
- 心跳超时与离线清理由连接断开触发；如需更严格的离线检测，可在客户端断网时关闭连接或在服务端定时巡检。
